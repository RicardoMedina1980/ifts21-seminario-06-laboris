️ Laboris DB — Simulador MySQL

Base de datos: laboris_db | Motor: MySQL (simulado con localStorage)

ESQUEMA DE LA BASE DE DATOS CREATE DATABASE laboris_db; USE laboris_db;
CREATE TABLE clientes (   id INT AUTO_INCREMENT PRIMARY KEY,   nombre VARCHAR(100) NOT NULL,   email VARCHAR(100) NOT NULL,   celular VARCHAR(20),   fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP );
CREATE TABLE expedientes (   id INT AUTO_INCREMENT PRIMARY KEY,   numero VARCHAR(20) UNIQUE NOT NULL,   caratula VARCHAR(200),   estado VARCHAR(50) NOT NULL,   etapa VARCHAR(100),   fecha DATE,   cliente_id INT REFERENCES clientes(id) );
CREATE TABLE consultas (   id INT AUTO_INCREMENT PRIMARY KEY,   nombre VARCHAR(100) NOT NULL,   email VARCHAR(100) NOT NULL,   celular VARCHAR(20),   tipo_consulta VARCHAR(50),   mensaje TEXT,   fecha_envio DATETIME DEFAULT CURRENT_TIMESTAMP );
